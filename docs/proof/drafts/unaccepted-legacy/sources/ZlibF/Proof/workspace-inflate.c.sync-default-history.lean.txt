/-- An early sync return still has a live initialized state. The temporary
four-byte block was freed before the caller regains control. -/
theorem good_tmpFreed {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} (good : Good h r zs z s) :
    Good (tmpFreed h) r zs z s := by
  have distinct := win_ne_buf good.fresh good.inv.win
  simpa [tmpFreed, syncBuf] using
    good_freeBuf (H := syncAlloc h) (nb := h.nextBlk) (good_syncAlloc good) distinct

/-- A sync call that returns before scanning leaves the complete default
invariant and bounded search count intact after freeing its local object. -/
theorem DefaultSyncJ.tmpFreed {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} (invariant : DefaultSyncJ h r zs z s) :
    DefaultSyncJ (tmpFreed h) r zs z s := by
  have allocated := default_state_syncAlloc invariant.toDefaultStateJ
  have localSize : (syncJunk h).size = 4 := by simp [syncJunk]
  refine ⟨?_, invariant.searchBound⟩
  change DefaultStateJ (freeWin (syncAlloc h) (.ref (.blk h.nextBlk) 0)) r zs z s
  exact allocated.free_local (syncAlloc_bytes_buf h) localSize


/-- Shifting out every whole byte of a normalized bit buffer leaves zero. -/
theorem shr8N_zero_of_bound (n : Nat) (x : UInt64)
    (bound : x.toNat < 2 ^ (8 * n)) : shr8N n x = 0 := by
  apply UInt64.toNat_inj.mp
  rw [shr8N_toNat, Nat.div_eq_of_lt bound]
  rfl


/-- With no input and fewer than one held byte, the actual sync call returns
BUF_ERROR and retains the complete default relation on its returned heap. -/
theorem inflateSync_default_buf_history {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} (reachable : StreamReachable .inflate h r)
    (invariant : DefaultSyncJ h r zs z s)
    (noInput : z.avail_in = 0) (fewBits : s.bits < 8)
    (arguments : StreamArguments h r (.inflateSync (some r))) :
    C.run (inflateSync (some r)) h = .ok (Z_BUF_ERROR, tmpFreed h) ∧
    DefaultSyncJ (tmpFreed h) r zs z s ∧
    StreamReachable .inflate (tmpFreed h) r := by
  have run := inflateSync_buf_error invariant.good noInput fewBits
  have executed : (Call.inflateSync (some r)).execute default h =
      .ok ((Z_BUF_ERROR, default), tmpFreed h) := by
    change inflateSync (some r) h = _ at run
    simp only [Call.execute, heapCall, run]
  have operation : StreamOperation .inflate r (.inflateSync (some r)) := by
    simp [StreamOperation]
  exact ⟨run, invariant.tmpFreed,
    StreamReachable.step reachable operation arguments executed⟩

/-- A resumed search that has not found the marker keeps the complete
default relation once the actual return supplies `Good` and the heap frame. -/
theorem DefaultStateJ.sync_resume_bad {h after : Heap} {r : ZStreamRef} {zs : Nat}
    {z stream : ZStream} {s : InflateState} {G : UInt32}
    (invariant : DefaultStateJ h r zs z s) (mode : s.mode = SYNC)
    (good : Good after r zs stream (syncHave s G))
    (platform : ZlibF.PlatformMemory.ByteObjectsFit after)
    (static : after.static = h.static) :
    DefaultStateJ after r zs stream (syncHave s G) := by
  have nextMode : (syncHave s G).mode = SYNC := by simp [syncHave, mode]
  refine ⟨⟨good, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, platform⟩,
    ?_, ?_, ?_⟩
  · intro active
    have inactive : ¬ DataTableMode SYNC := by unfold DataTableMode; decide
    exact False.elim (inactive (by simpa [nextMode] using active))
  · apply dataPendingInvariant_idle
    · simpa [syncHave] using invariant.pending.sane
    · rw [nextMode]
      decide
  · intro active
    rw [nextMode] at active
    contradiction
  · intro active
    rw [nextMode] at active
    contradiction
  · simpa [syncHave] using invariant.held
  · simpa [syncHave] using invariant.window
  · simpa [syncHave] using invariant.wrapper
  · simpa [syncHave] using invariant.header
  · right
    rw [nextMode]
    decide
  · intro active
    rcases active with check | length
    · rw [nextMode] at check
      contradiction
    · rw [nextMode] at length
      contradiction
  · change after.static = _
    rw [static]
    exact invariant.globals
  · simpa [syncHave] using invariant.requestedBits
  · simpa [syncHave] using invariant.requestedWrap
  · exact Or.inr (Or.inr (Or.inr (Or.inl nextMode)))

/-- The resumed-search data-error heap changes only the stream, inflate
record and the bounded temporary byte block. -/
theorem platform_sync_resume_bad {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} {N G : UInt32}
    (platform : ZlibF.PlatformMemory.ByteObjectsFit h) :
    ZlibF.PlatformMemory.ByteObjectsFit
      (freeWin (setI (r.put (syncAlloc h) (syncAdvance z N)) zs (syncHave s G))
        (syncBuf h)) := by
  have allocated := default_platform_syncAlloc platform
  have streamed := allocated.put_stream r (syncAdvance z N)
  have stated := streamed.with_istates
    (upd (r.put (syncAlloc h) (syncAdvance z N)).istates zs (some (syncHave s G)))
  have freed := platform_free_local stated h.nextBlk
  simpa only [setI, syncBuf] using freed

/-- Restoring at most four held bytes rewrites only the bounded temporary
buffer before the failed first search returns. -/
theorem platform_sync_first_bad {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} {N G : UInt32}
    (platform : ZlibF.PlatformMemory.ByteObjectsFit h) :
    ZlibF.PlatformMemory.ByteObjectsFit
      (freeWin
        (setI
          (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
            (syncAdvance z N)) zs (syncHave (syncFirstState s) G))
        (syncBuf h)) := by
  have initial := default_platform_syncAlloc platform
  have bounded : (fillBuf (syncJunk h) (syncHold0 s) (syncNB s)).size <
      ZlibF.PlatformMemory.byteObjectLimit := by
    rw [fillBuf_size]
    change 4 < 2 ^ 63
    decide
  have written := initial.with_bytes (block := h.nextBlk) bounded
  have restored : ZlibF.PlatformMemory.ByteObjectsFit
      (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h)) := by
    apply written.of_storage_eq
    all_goals rfl
  have streamed := restored.put_stream r (syncAdvance z N)
  have stated := streamed.with_istates
    (upd (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
      (syncAdvance z N)).istates zs (some (syncHave (syncFirstState s) G)))
  have freed := platform_free_local stated h.nextBlk
  simpa only [setI, syncBuf] using freed

/-- Default-wrapper histories have only the unread-header or zlib-header
flag values. Either value satisfies the TYPE-phase clause after sync reset. -/
theorem syncOkState_default_phase (zs : Nat) (s : InflateState)
    (flags : s.flags = -1 ∨ s.flags = 0) :
    (syncOkState zs s).wrap = 0 ∨ (syncOkState zs s).flags = 0 := by
  rcases flags with unread | zlib
  · left
    simp [syncOkState, syncC, resetKeepState, unread] <;> split <;> rfl
  · right
    simp [syncOkState, syncC, resetKeepState, zlib] <;> split <;> rfl

/-- The actual default init execution returns the unread-header flag. -/
theorem inflate_default_success_flags {h after : Heap} {r : ZStreamRef} {version : BPtr}
    (domain : InitDomain h r version) (allocation : h.allocOk h.nalloc = true)
    (run : C.run (inflateInit_ (some r) version 112) h = .ok (Z_OK, after))
    {z : ZStream} {s : InflateState} (good : Good after r h.nextBlk z s) :
    s.flags = -1 := by
  obtain ⟨fresh, _, _, validVersion, before, get, allocators, _⟩ := domain
  obtain ⟨nonnull, first⟩ :=
    ZlibF.Spec.Public.Proofs.InflateInit.version_value_read validVersion
  have expected : C.run (inflateInit_ (some r) version 112) h =
      .ok (Z_OK, initOkHeap h r before 5 15) := by
    change inflateInit2_ (some r) 15 version 112 h = _
    exact inflateInit2_success get fresh allocation allocators.1 allocators.2
      nonnull first (by decide +kernel)
  have heapEq : after = initOkHeap h r before 5 15 :=
    congrArg Prod.snd (Except.ok.inj (run.symm.trans expected))
  have stateEq : s = resetKeepState h.nextBlk
      (resetPre (reset2Pre 5 15 (initState r))) := by
    apply Option.some.inj
    rw [← good.live.hs, heapEq]
    simp [initOkHeap]
  rw [stateEq]
  rfl

/-- The default initializer establishes both the decoder/search relation
and the flag fact required by a later successful sync reset. -/
theorem inflate_default_sync_flags_initial (h : Heap) (r : ZStreamRef) (version : BPtr)
    (domain : InitDomain h r version) (platform : ZlibF.PlatformMemory.ByteObjectsFit h)
    (allocation : h.allocOk h.nalloc = true) :
    ∃ after z s, C.run (inflateInit_ (some r) version 112) h = .ok (Z_OK, after) ∧
      DefaultSyncJ after r h.nextBlk z s ∧ s.flags = -1 ∧
      StreamReachable .inflate after r := by
  obtain ⟨after, z, s, run, invariant, reachable⟩ :=
    inflate_default_sync_initial h r version domain platform allocation
  exact ⟨after, z, s, run, invariant,
    inflate_default_success_flags domain allocation run invariant.good, reachable⟩

/-- The marker-found reset re-establishes the default TYPE invariant when
the admitted default-wrapper flag value is carried from its history. -/
theorem DefaultStateJ.sync_done {h after : Heap} {r : ZStreamRef} {zs : Nat}
    {z stream : ZStream} {s : InflateState}
    (invariant : DefaultStateJ h r zs z s)
    (flags : s.flags = -1 ∨ s.flags = 0)
    (good : Good after r zs stream (syncOkState zs s))
    (platform : ZlibF.PlatformMemory.ByteObjectsFit after)
    (static : after.static = h.static) :
    DefaultStateJ after r zs stream (syncOkState zs s) := by
  have nonzero : s.reqbits ≠ 0 := by rw [invariant.requestedBits]; decide
  have typeMode : (syncOkState zs s).mode = TYPE := by simp [syncOkState]
  have sameBits : s.wbits = s.reqbits := invariant.window.trans invariant.requestedBits.symm
  have pre : resetPre s =
      { s with wrap := s.reqwrap, wsize := 0, whave := 0, wnext := 0 } := by
    simp only [resetPre, sameBits, bne_self_eq_false, Bool.false_eq_true, ite_false]
  obtain ⟨wbits, _, _, _, _, reqbits, _, _, _, _, _⟩ := syncC_field zs s
  refine ⟨⟨good, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, ?_, platform⟩,
    ?_, ?_, ?_⟩
  · intro active
    have inactive : ¬ DataTableMode TYPE := by unfold DataTableMode; decide
    exact False.elim (inactive (by simpa [typeMode] using active))
  · apply dataPendingInvariant_idle
    · rw [syncOkState_ne0 nonzero]
      simp only [InflateState.sane]
      simp [syncC, resetKeepState] <;> split <;> rfl
    · rw [typeMode]
      decide
  · intro active
    rw [typeMode] at active
    contradiction
  · intro active
    rw [typeMode] at active
    contradiction
  · have zero : (syncOkState zs s).hold = 0 := by
      rw [syncOkState_ne0 nonzero]
      simp [syncC, resetKeepState] <;> split <;> rfl
    have bits : (syncOkState zs s).bits = 0 := by
      rw [syncOkState_ne0 nonzero]
      exact (syncC_field zs s).2.2.2.2.2.2.2.1
    rw [zero, bits]
    change 0 < 1
    decide
  · rw [syncOkState_ne0 nonzero, wbits]
    exact invariant.requestedBits
  · rcases flags with unread | zlib
    · left
      simp [syncOkState, syncC, resetKeepState, unread] <;> split <;> rfl
    · rcases invariant.requestedWrap with one | five
      · right; left
        rw [syncOkState_ne0 nonzero]
        simp [syncC, resetKeepState, pre, zlib, one] <;> decide
      · right; left
        rw [syncOkState_ne0 nonzero]
        simp [syncC, resetKeepState, pre, zlib, five] <;> decide
  · rw [syncOkState_ne0 nonzero]
    simp [syncC, resetKeepState] <;> split <;> rfl
  · right
    rw [typeMode]
    decide
  · intro active
    rcases active with check | length
    · rw [typeMode] at check
      contradiction
    · rw [typeMode] at length
      contradiction
  · change after.static = _
    rw [static]
    exact invariant.globals
  · rw [syncOkState_ne0 nonzero, reqbits]
    exact invariant.requestedBits
  · have saved : (syncOkState zs s).reqwrap = s.reqwrap := by
      rw [syncOkState_ne0 nonzero]
      simp [syncC, resetKeepState, pre] <;> split <;> rfl
    rw [saved]
    exact invariant.requestedWrap
  · exact Or.inr (Or.inr (Or.inr (Or.inr (syncOkState_default_phase zs s flags))))

/-- In the default window configuration, the successful sync reset keeps
the existing window and performs only stream and decoder-record writes. -/
theorem syncOkHeap_default_eq (H : Heap) (r : ZStreamRef) (zs : Nat)
    (z : ZStream) (s : InflateState) (sameBits : s.wbits = s.reqbits)
    (requested : s.reqbits = 15) :
    syncOkHeap H r zs z s =
      setI (r.put H (syncOkStream z s)) zs (syncOkState zs s) := by
  have reset : resetHeap H s = H := by
    simp [resetHeap, sameBits]
  have nonzero : (syncC zs s).wbits ≠ 0 := by
    rw [(syncC_field zs s).1, requested]
    decide
  simp [syncOkHeap, reset, nonzero]

/-- Consequently successful reset preserves the platform bound and static
tables whenever the pre-reset heap satisfies them. -/
theorem platform_syncOkHeap_default {H : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState}
    (platform : ZlibF.PlatformMemory.ByteObjectsFit H)
    (sameBits : s.wbits = s.reqbits) (requested : s.reqbits = 15) :
    ZlibF.PlatformMemory.ByteObjectsFit (syncOkHeap H r zs z s) := by
  rw [syncOkHeap_default_eq H r zs z s sameBits requested]
  exact (platform.put_stream r (syncOkStream z s)).with_istates _

theorem static_syncOkHeap_default (H : Heap) (r : ZStreamRef) (zs : Nat)
    (z : ZStream) (s : InflateState) (sameBits : s.wbits = s.reqbits)
    (requested : s.reqbits = 15) :
    (syncOkHeap H r zs z s).static = H.static := by
  rw [syncOkHeap_default_eq H r zs z s sameBits requested]
  cases r <;> simp [setI, ZStreamRef.put] <;> split <;> rfl

/-- If a continued sync search does not find the marker, its actual
DATA_ERROR return preserves the complete default invariant and history. -/
theorem inflateSync_resume_default_bad_history {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} {bytes seen : List UInt8}
    (reachable : StreamReachable .inflate h r)
    (invariant : DefaultSyncJ h r zs z s) (allocator : DefAlloc z)
    (mode : s.mode = SYNC) (progress : ¬ (z.avail_in = 0 ∧ s.bits < 8))
    (input : SyncIn h z.next_in z.avail_in bytes)
    (atSpec : SyncAt s.«have» seen)
    (arguments : StreamArguments h r (.inflateSync (some r))) :
    ∃ N G, N.toNat ≤ z.avail_in.toNat ∧
      (G ≠ 4 → ∃ after,
        C.run (inflateSync (some r)) h = .ok (Z_DATA_ERROR, after) ∧
        DefaultSyncJ after r zs (syncAdvance z N) (syncHave s G) ∧
        StreamReachable .inflate after r) := by
  obtain ⟨N, G, used, _, rest, bad, _⟩ :=
    inflateSync_resume invariant.good allocator.2 mode progress input atSpec
  refine ⟨N, G, used, ?_⟩
  intro missed
  obtain ⟨after, run, shape, good⟩ := bad missed
  subst after
  have platform := platform_sync_resume_bad (r := r) (zs := zs)
    (z := z) (s := s) (N := N) (G := G) invariant.platform
  have static := static_sync_resume_bad h r zs z s N G
  have state := invariant.toDefaultStateJ.sync_resume_bad mode good platform static
  have bound : (syncHave s G).mode = SYNC → (syncHave s G).«have».toNat ≤ 4 := by
    intro _
    have count : G.toNat ≤ 4 := Nat.le_of_lt (rest missed).2.1
    simpa [syncHave] using count
  have executed : (Call.inflateSync (some r)).execute default h =
      .ok ((Z_DATA_ERROR, default),
        freeWin (setI (r.put (syncAlloc h) (syncAdvance z N)) zs (syncHave s G))
          (syncBuf h)) := by
    change inflateSync (some r) h = _ at run
    simp only [Call.execute, heapCall, run]
  have operation : StreamOperation .inflate r (.inflateSync (some r)) := by
    simp [StreamOperation]
  exact ⟨_, run, ⟨state, bound⟩,
    StreamReachable.step reachable operation arguments executed⟩

/-- A first sync search that misses the marker also returns with the full
default relation and a bounded successor search count. -/
theorem inflateSync_first_default_bad_history {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} {bytes : List UInt8}
    (reachable : StreamReachable .inflate h r)
    (invariant : DefaultSyncJ h r zs z s) (allocator : DefAlloc z)
    (mode : s.mode ≠ SYNC) (progress : ¬ (z.avail_in = 0 ∧ s.bits < 8))
    (input : SyncIn h z.next_in z.avail_in bytes)
    (arguments : StreamArguments h r (.inflateSync (some r))) :
    ∃ N G, N.toNat ≤ z.avail_in.toNat ∧
      (G ≠ 4 → ∃ after,
        C.run (inflateSync (some r)) h = .ok (Z_DATA_ERROR, after) ∧
        DefaultSyncJ after r zs (syncAdvance z N) (syncHave (syncFirstState s) G) ∧
        StreamReachable .inflate after r) := by
  obtain ⟨N, G, used, _, rest, bad, _⟩ :=
    inflateSync_first invariant.good allocator.2 mode progress input
  refine ⟨N, G, used, ?_⟩
  intro missed
  obtain ⟨after, run, shape, good⟩ := bad missed
  subst after
  have platform := platform_sync_first_bad (r := r) (zs := zs)
    (z := z) (s := s) (N := N) (G := G) invariant.platform
  have static := static_sync_first_bad h r zs z s N G
  have state := invariant.toDefaultStateJ.sync_first_bad good platform static
  have bound : (syncHave (syncFirstState s) G).mode = SYNC →
      (syncHave (syncFirstState s) G).«have».toNat ≤ 4 := by
    intro _
    have count : G.toNat ≤ 4 := Nat.le_of_lt (rest missed).2.1
    simpa [syncHave] using count
  have executed : (Call.inflateSync (some r)).execute default h =
      .ok ((Z_DATA_ERROR, default),
        freeWin
          (setI
            (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
              (syncAdvance z N)) zs (syncHave (syncFirstState s) G))
          (syncBuf h)) := by
    change inflateSync (some r) h = _ at run
    simp only [Call.execute, heapCall, run]
  have operation : StreamOperation .inflate r (.inflateSync (some r)) := by
    simp [StreamOperation]
  exact ⟨_, run, ⟨state, bound⟩,
    StreamReachable.step reachable operation arguments executed⟩

/-- For a public raw input block, every actual sync return is safe and
leaves a bounded search count if the stream remains in SYNC. -/
theorem inflateSync_default_raw_history {h : Heap} {r : ZStreamRef} {zs b off : Nat}
    {z : ZStream} {s : InflateState} (reachable : StreamReachable .inflate h r)
    (invariant : DefaultSyncJ h r zs z s) (allocator : DefAlloc z)
    (pointer : z.next_in = .ref (.blk b) off)
    (arguments : StreamArguments h r (.inflateSync (some r))) :
    ∃ rc after stream state,
      C.run (inflateSync (some r)) h = .ok (rc, after) ∧
      Good after r zs stream state ∧
      (state.mode = SYNC → state.«have».toNat ≤ 4) ∧
      StreamReachable .inflate after r := by
  have argumentCopy := arguments
  obtain ⟨given, givenGet, range⟩ := argumentCopy
  have same : given = z :=
    Option.some.inj (givenGet.symm.trans invariant.good.live.hz)
  subst given
  rw [pointer] at range
  obtain ⟨bytes, input⟩ := syncIn_of_callerRange_blk (len := z.avail_in) range
  have inputZ : SyncIn h z.next_in z.avail_in bytes := by
    simpa only [pointer] using input
  have nextOf : ∀ (rc : Int32) (after : Heap),
      C.run (inflateSync (some r)) h = .ok (rc, after) →
      StreamReachable .inflate after r := by
    intro rc after run
    have executed : (Call.inflateSync (some r)).execute default h =
        .ok ((rc, default), after) := by
      change inflateSync (some r) h = _ at run
      simp only [Call.execute, heapCall, run]
    have operation : StreamOperation .inflate r (.inflateSync (some r)) := by
      simp [StreamOperation]
    exact StreamReachable.step reachable operation arguments executed
  by_cases short : z.avail_in = 0 ∧ s.bits < 8
  · have run := inflateSync_buf_error invariant.good short.1 short.2
    exact ⟨Z_BUF_ERROR, tmpFreed h, z, s, run, good_tmpFreed invariant.good,
      invariant.searchBound, nextOf Z_BUF_ERROR (tmpFreed h) run⟩
  by_cases inSync : s.mode = SYNC
  · obtain ⟨seen, atSpec⟩ := invariant.syncAt inSync
    obtain ⟨N, G, used, found, rest, bad, done⟩ :=
      inflateSync_resume invariant.good allocator.2 inSync short inputZ atSpec
    by_cases complete : G = 4
    · obtain ⟨after, run, _, good⟩ := done complete
      have bound : (syncOkState zs (syncHave s G)).mode = SYNC →
          (syncOkState zs (syncHave s G)).«have».toNat ≤ 4 := by
        intro mode
        have different : TYPE ≠ SYNC := by decide
        exact False.elim (different (by simpa [syncOkState] using mode))
      exact ⟨Z_OK, after, syncOkStream (syncAdvance z N) (syncHave s G),
        syncOkState zs (syncHave s G), run, good, bound, nextOf Z_OK after run⟩
    · obtain ⟨after, run, _, good⟩ := bad complete
      have count : G.toNat ≤ 4 := Nat.le_of_lt (rest complete).2.1
      have bound : (syncHave s G).mode = SYNC → (syncHave s G).«have».toNat ≤ 4 := by
        intro _
        simpa [syncHave] using count
      exact ⟨Z_DATA_ERROR, after, syncAdvance z N, syncHave s G, run, good,
        bound, nextOf Z_DATA_ERROR after run⟩
  · obtain ⟨N, G, used, found, rest, bad, done⟩ :=
      inflateSync_first invariant.good allocator.2 inSync short inputZ
    by_cases complete : G = 4
    · obtain ⟨after, run, _, good⟩ := done complete
      have bound : (syncOkState zs (syncHave (syncFirstState s) G)).mode = SYNC →
          (syncOkState zs (syncHave (syncFirstState s) G)).«have».toNat ≤ 4 := by
        intro mode
        have different : TYPE ≠ SYNC := by decide
        exact False.elim (different (by simpa [syncOkState] using mode))
      exact ⟨Z_OK, after,
        syncOkStream (syncAdvance z N) (syncHave (syncFirstState s) G),
        syncOkState zs (syncHave (syncFirstState s) G), run, good, bound,
        nextOf Z_OK after run⟩
    · obtain ⟨after, run, _, good⟩ := bad complete
      have count : G.toNat ≤ 4 := Nat.le_of_lt (rest complete).2.1
      have bound : (syncHave (syncFirstState s) G).mode = SYNC →
          (syncHave (syncFirstState s) G).«have».toNat ≤ 4 := by
        intro _
        simpa [syncHave] using count
      exact ⟨Z_DATA_ERROR, after, syncAdvance z N,
        syncHave (syncFirstState s) G, run, good, bound,
        nextOf Z_DATA_ERROR after run⟩

end Zlib.Inflate.Proofs
