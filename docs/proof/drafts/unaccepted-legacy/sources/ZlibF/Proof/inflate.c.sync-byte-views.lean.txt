/-- A non-null public caller range supplies the bytes of any represented
byte view, including literals and static tables. -/
theorem syncBytes_of_callerRange {h : Heap} {r : ZStreamRef}
    {p : BPtr} {len : UInt32}
    (range : CallerRange h r p len.toNat false) (nonnull : p ≠ .null) :
    ∃ bs, Bytes h p len.toNat bs := by
  rcases range with null | ⟨bytes, _, _⟩
  · exact False.elim (nonnull null)
  · exact bytes

/-- Fresh local allocation preserves every already readable public view. -/
theorem syncPublicBytes_alloc {h : Heap} {p : BPtr} {len : UInt32}
    {bs : List UInt8} (fresh : Fresh h)
    (input : Bytes h p len.toNat bs) :
    Bytes (syncAlloc h) p len.toNat bs := by
  rcases input with null | ⟨ref, off, array, pointer, get, fit, contents⟩
  · exact Or.inl null
  · exact Or.inr ⟨ref, off, array, pointer,
      (byte_view_syncAlloc fresh get).1, fit, contents⟩

/-- Restoring the held bits writes only the newly allocated four-byte block. -/
theorem syncPublicBytes_first {h : Heap} {p : BPtr} {len : UInt32}
    {bs : List UInt8} (fresh : Fresh h) (zs : Nat) (s : InflateState)
    (input : Bytes h p len.toNat bs) :
    Bytes (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
      p len.toNat bs := by
  rcases input with null | ⟨ref, off, array, pointer, get, fit, contents⟩
  · exact Or.inl null
  · have get' : (Space.get (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h)) ref :
        Option (Array UInt8)) = some array := by
      have allocated := (byte_view_syncAlloc fresh get).1
      cases ref with
      | blk block =>
        have different : block ≠ h.nextBlk := by
          intro same
          subst block
          have absent := (fresh_dead fresh (Nat.le_refl h.nextBlk)).1
          change h.bytes h.nextBlk = some array at get
          rw [get] at absent
          cases absent
        simpa [Space.get, syncFirstHeap, upd_ne _ _ _ _ different] using allocated
      | str index => exact allocated
      | dstate_depth index => exact allocated
      | st_dist_code => exact allocated
      | st_length_code => exact allocated
    exact Or.inr ⟨ref, off, array, pointer, get', fit, contents⟩

/-- A positive-length null input faults on the first scanned byte whenever
the marker has not already been found. -/
theorem syncsearch_null_positive_fault (h : Heap) (g len : UInt32)
    (positive : len ≠ 0) (scanning : g < 4) :
    syncsearch g .null len h = .error (.ub "null dereference") := by
  have less : (0 : UInt32) < len := by
    rw [UInt32.lt_iff_toNat_lt]
    simp only [UInt32.toNat_zero]
    have nz : len.toNat ≠ 0 := by
      intro equal
      exact positive (UInt32.toNat_inj.mp (by simpa using equal))
    omega
  unfold syncsearch cwhile
  have first : fuel = (fuel - 1) + 1 := by unfold fuel; omega
  rw [first]
  simp [cwhile.go, less, scanning, index8, Ptr.index, Ptr.add, Ptr.load]

/-- Once `have` reaches four, the loop reads no input, even when the caller
supplies a null pointer and a positive count. -/
theorem syncsearch_already_found (h : Heap) (g len : UInt32) (p : BPtr)
    (found : ¬ g < 4) :
    syncsearch g p len h = .ok ((0, g), h) := by
  unfold syncsearch cwhile
  have first : fuel = (fuel - 1) + 1 := by unfold fuel; omega
  rw [first]
  simp [cwhile.go, found]

/-- A previously found marker permits a null input with positive count:
the search consumes zero bytes and the reset succeeds. -/
theorem syncTail_null_found {H : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState}
    (good : Good H r zs z s) (free : z.zfree = .fn "zcfree")
    (null : z.next_in = .null) (found : s.«have» = 4) :
    syncTail (some r) (some (.tab zs)) H =
      .ok (Z_OK,
        syncOkHeap (setI (r.put H (syncAdvance z 0)) zs (syncHave s 4))
          r zs (syncAdvance z 0) (syncHave s 4)) := by
  have live := good.live
  have scan : syncsearch s.«have» .null z.avail_in H =
      .ok ((0, 4), H) := by
    rw [found]
    exact syncsearch_already_found H 4 z.avail_in .null (by decide)
  have searched := good_syncSearch good 0 4
  have finished := syncDone_run searched
    (by simpa [syncAdvance] using free)
  have advance : Ptr.add (.null : BPtr) 0
      (setI H zs { s with «have» := 4 }) =
      .ok (.null, setI H zs { s with «have» := 4 }) :=
    Ptr.add_null_zero
  unfold syncTail
  isimp [live.hz, live.hs, scan, null, advance,
    syncAdvance, syncHave]
  simpa [syncAdvance, syncAdv, syncHave, null] using finished

/-- A resumed search with a positive null input and an unfinished marker
cannot return successfully: its next byte read faults. -/
theorem syncTail_null_positive_fault {H : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} (good : Good H r zs z s)
    (positive : z.avail_in ≠ 0) (null : z.next_in = .null)
    (scanning : s.«have» < 4) :
    syncTail (some r) (some (.tab zs)) H =
      .error (.ub "null dereference") := by
  have scan : syncsearch s.«have» .null z.avail_in H =
      .error (.ub "null dereference") :=
    syncsearch_null_positive_fault H s.«have» z.avail_in positive scanning
  unfold syncTail
  isimp [good.live.hz, good.live.hs, null, scan]

/-- The complete resumed call has no successful history step in that case. -/
theorem inflateSync_resume_null_positive_fault {h : Heap} {r : ZStreamRef}
    {zs : Nat} {z : ZStream} {s : InflateState}
    (good : Good h r zs z s) (mode : s.mode = SYNC)
    (positive : z.avail_in ≠ 0) (null : z.next_in = .null)
    (scanning : s.«have» < 4) :
    C.run (inflateSync (some r)) h =
      .error (.ub "null dereference") := by
  have allocated := good_syncAlloc good
  have tail := syncTail_null_positive_fault allocated positive null scanning
  have progress : ¬ (z.avail_in = 0 ∧ s.bits < 8) :=
    fun short => positive short.1
  show inflateSync (some r) h = _
  rw [inflateSync_split]
  isimp [allocLocalB_run h good.fresh, stateCheck_live allocated.live,
    allocated.live.zst, good.live.hz, good.live.hs, progress, mode, tail]

/-- If the held bytes complete the marker, the first sync call succeeds
without reading a positive-length null caller input. -/
theorem inflateSync_first_null_found {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState}
    (good : Good h r zs z s) (free : z.zfree = .fn "zcfree")
    (mode : s.mode ≠ SYNC) (positive : z.avail_in ≠ 0)
    (null : z.next_in = .null)
    (found : (syncFirstState s).«have» = 4) :
    C.run (inflateSync (some r)) h =
      .ok (Z_OK,
        freeWin
          (syncOkHeap
            (setI
              (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
                (syncAdvance z 0)) zs (syncHave (syncFirstState s) 4))
            r zs (syncAdvance z 0) (syncHave (syncFirstState s) 4))
          (syncBuf h)) := by
  have fresh := good.fresh
  have allocated := good_syncAlloc good
  have distinct := win_ne_buf fresh good.inv.win
  have localGet := syncAlloc_bytes_buf h
  have localSize : (syncJunk h).size = 4 := by simp [syncJunk]
  have firstGood := good_firstState allocated localGet distinct
  have firstRun := syncFirst_run
    (H0 := syncAlloc h) (zs := zs) (nb := h.nextBlk)
    (st := s) (A := syncJunk h)
    (by simpa using good.live.hs) localGet localSize (syncAlloc_ro_buf h)
    good.inv.bits
  have tail := syncTail_null_found firstGood free null found
  have different : ∀ w,
      (syncHave (syncFirstState s) 4).window = .ref (.blk w) 0 →
        w ≠ h.nextBlk :=
    fun w window => distinct w
      (by simpa [syncHave, syncFirstState] using window)
  have bytes := syncOkHeap_bytes_ne
    (setI
      (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
        (syncAdvance z 0)) zs (syncHave (syncFirstState s) 4))
    r zs (syncAdvance z 0) (syncHave (syncFirstState s) 4) different
  have finalLocal :
      (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h)).bytes h.nextBlk =
      some (fillBuf (syncJunk h) (syncHold0 s) (syncNB s)) := by
    simp [syncFirstHeap]
  have finalRo :
      (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h)).ro h.nextBlk = false := by
    simp [syncFirstHeap, syncAlloc_ro_buf]
  have progress : ¬ (z.avail_in = 0 ∧ s.bits < 8) :=
    fun short => positive short.1
  show inflateSync (some r) h = _
  rw [inflateSync_split]
  isimp [allocLocalB_run h fresh, stateCheck_live allocated.live,
    allocated.live.zst, good.live.hz, good.live.hs, progress, mode,
    syncBuf, firstRun, tail,
    freeBuf_run
      (H := syncOkHeap
        (setI
          (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
            (syncAdvance z 0)) zs (syncHave (syncFirstState s) 4))
        r zs (syncAdvance z 0) (syncHave (syncFirstState s) 4))
      (nb := h.nextBlk)
      (a := fillBuf (syncJunk h) (syncHold0 s) (syncNB s))
      (by rw [bytes.1]; simpa using finalLocal)
      (by rw [bytes.2]; simpa using finalRo)]

/-- The generic byte-view packet has exactly the advertised input length. -/
theorem syncPublicBytes_length {h : Heap} {p : BPtr} {len : UInt32}
    {bs : List UInt8} (input : Bytes h p len.toNat bs) :
    bs.length = len.toNat := by
  rcases input with ⟨rfl, zero, rfl⟩ | ⟨ref, off, array, rfl, get, fit, rfl⟩
  · simpa [zero]
  · simp only [List.length_take, List.length_drop, Array.length_toList]
    omega

/-- The scanner's checked execution and first-match result hold for every
public byte view, with no change to the existing raw-block specification. -/
theorem syncsearch_spec_bytes {h : Heap} {p : BPtr} {len : UInt32}
    {bs : List UInt8} (input : Bytes h p len.toNat bs)
    {g : UInt32} {w : List UInt8} (seen : SyncSt g w) :
    ∃ N G, syncsearch g p len h = .ok ((N, G), h) ∧ N.toNat ≤ len.toNat ∧
      (G = 4 → 0 < N.toNat ∧ syncPat <:+ w ++ bs.take N.toNat ∧
        ∀ L', L' < N.toNat → ¬ syncPat <:+: w ++ bs.take L') ∧
      (G ≠ 4 → N = len ∧ SyncSt G (w ++ bs)) := by
  have size := syncPublicBytes_length input
  rcases input with ⟨rfl, zero, rfl⟩ |
    ⟨ref, off, array, rfl, get, fit, rfl⟩
  · have lenZero : len = 0 := UInt32.toNat_inj.mp (by simpa using zero)
    subst len
    refine ⟨0, g, syncsearch_null g h, by simp, ?_, ?_⟩
    · intro found
      have notDone := seen.1
      rw [found] at notDone
      exact absurd notDone (by decide)
    · intro _
      exact ⟨rfl, by simpa using seen⟩
  · let bytes := (array.toList.drop off).take len.toNat
    have sizeBytes : bytes.length = len.toNat := by
      dsimp [bytes]
      simpa only [List.length_take, List.length_drop, Array.length_toList] using size
    refine ⟨UInt32.ofNat (syncScan g bytes).2, (syncScan g bytes).1,
      syncsearch_run g get fit, ?_⟩
    obtain ⟨used, found, missed⟩ := syncScan_spec seen bytes rfl
    have count : (UInt32.ofNat (syncScan g bytes).2).toNat =
        (syncScan g bytes).2 := by
      have small := len.toNat_lt
      rw [UInt32.toNat_ofNat_of_lt' (by simp [UInt32.size]; omega)]
    rw [count]
    refine ⟨by omega, found, fun notFound => ?_⟩
    obtain ⟨equal, state⟩ := missed notFound
    refine ⟨?_, state⟩
    apply UInt32.toNat_inj.mp
    rw [count]
    omega

/-- An already-found marker also stays found for any byte view. -/
theorem syncsearch_at_bytes {h : Heap} {p : BPtr} {len : UInt32}
    {bs : List UInt8} (input : Bytes h p len.toNat bs)
    {g : UInt32} {w : List UInt8} (seen : SyncAt g w) :
    ∃ N G, syncsearch g p len h = .ok ((N, G), h) ∧ N.toNat ≤ len.toNat ∧
      (G = 4 → syncPat <:+: w ++ bs.take N.toNat ∧
        ∀ L', L' < N.toNat → ¬ syncPat <:+: w ++ bs.take L') ∧
      (G ≠ 4 → N = len ∧ SyncSt G (w ++ bs)) := by
  rcases seen with state | ⟨rfl, pattern⟩
  · obtain ⟨N, G, run, used, found, missed⟩ :=
      syncsearch_spec_bytes input state
    refine ⟨N, G, run, used, ?_, missed⟩
    intro complete
    obtain ⟨_, suffix, first⟩ := found complete
    exact ⟨List.IsSuffix.isInfix suffix, first⟩
  · rcases input with ⟨rfl, zero, rfl⟩ |
      ⟨ref, off, array, rfl, get, fit, bytes⟩
    · have lenZero : len = 0 := UInt32.toNat_inj.mp (by simpa using zero)
      subst len
      exact ⟨0, 4, syncsearch_null 4 h, by simp,
        fun _ => ⟨by simpa using pattern, fun L' before => by simp at before⟩,
        fun notFound => absurd rfl notFound⟩
    · have run := syncsearch_run (r := ref) (h := h) (a := array)
        (o := off) (len := len) 4 get fit
      rw [syncScan_ge (by decide)] at run
      refine ⟨0, 4, by simpa using run, by simp, ?_, ?_⟩
      · intro _
        exact ⟨by simpa using pattern, fun L' before => by simp at before⟩
      · intro notFound
        exact absurd rfl notFound

/-- Advancing an input pointer uses the same byte view in the heap after the
scalar `have` update. -/
theorem syncAdd_run_bytes {H H' : Heap} {p : BPtr} {len N : UInt32}
    {bs : List UInt8} (input : Bytes H p len.toNat bs)
    (used : N.toNat ≤ len.toNat)
    (same : ∀ ref : BRef,
      (Space.get H' ref : Option (Array UInt8)) = Space.get H ref) :
    Ptr.add p (N.toNat : Int) H' = .ok (syncAdv p N.toNat, H') := by
  rcases input with ⟨rfl, zero, _⟩ | ⟨ref, off, array, rfl, get, fit, _⟩
  · have empty : N.toNat = 0 := by omega
    rw [empty]
    rfl
  · have get' : (Space.get H' ref : Option (Array UInt8)) = some array := by
      rw [same]
      exact get
    exact Ptr.add_nat_ok get' (by omega)

/-- The actual search tail works for every represented public byte view.
This is the generic companion to the raw-block tail theorem. -/
theorem syncTail_run_bytes {H : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} {bs w : List UInt8}
    (good : Good H r zs z s) (free : z.zfree = .fn "zcfree")
    (input : Bytes H z.next_in z.avail_in.toNat bs)
    (seen : SyncAt s.«have» w) :
    ∃ N G, N.toNat ≤ z.avail_in.toNat ∧
      (G = 4 → syncPat <:+: w ++ bs.take N.toNat ∧
        ∀ L', L' < N.toNat → ¬ syncPat <:+: w ++ bs.take L') ∧
      (G ≠ 4 → N = z.avail_in ∧ SyncSt G (w ++ bs)) ∧
      Good (setI (r.put H (syncAdvance z N)) zs (syncHave s G)) r zs
        (syncAdvance z N) (syncHave s G) ∧
      (G ≠ 4 → syncTail (some r) (some (.tab zs)) H =
        .ok (Z_DATA_ERROR,
          setI (r.put H (syncAdvance z N)) zs (syncHave s G))) ∧
      (G = 4 → syncTail (some r) (some (.tab zs)) H =
        .ok (Z_OK,
          syncOkHeap (setI (r.put H (syncAdvance z N)) zs (syncHave s G))
            r zs (syncAdvance z N) (syncHave s G))) := by
  have live := good.live
  obtain ⟨N, G, scan, used, found, missed⟩ := syncsearch_at_bytes input seen
  have searched := good_syncSearch good N G
  refine ⟨N, G, used, found, missed, searched, fun notFound => ?_, fun complete => ?_⟩
  · unfold syncTail
    isimp [live.hz, live.hs, scan,
      syncAdd_run_bytes input used
        (H' := setI H zs { s with «have» := G })
        (byte_view_get_setI H zs { s with «have» := G }),
      syncAdvance, syncHave, notFound]
  · subst G
    unfold syncTail
    have finished := syncDone_run searched (by simpa [syncAdvance] using free)
    isimp [live.hz, live.hs, scan,
      syncAdd_run_bytes input used
        (H' := setI H zs { s with «have» := 4 })
        (byte_view_get_setI H zs { s with «have» := 4 })]
    simpa [syncAdvance, syncHave] using finished

/-- A continued synchronization call reads any represented public byte
view. The actual C execution and returned heap agree with the raw case. -/
theorem inflateSync_resume_bytes {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} {bs seen : List UInt8}
    (good : Good h r zs z s) (free : z.zfree = .fn "zcfree")
    (mode : s.mode = SYNC) (progress : ¬ (z.avail_in = 0 ∧ s.bits < 8))
    (input : Bytes h z.next_in z.avail_in.toNat bs)
    (atSpec : SyncAt s.«have» seen) :
    ∃ N G, N.toNat ≤ z.avail_in.toNat ∧
      (G = 4 → syncPat <:+: seen ++ bs.take N.toNat ∧
        ∀ L', L' < N.toNat → ¬ syncPat <:+: seen ++ bs.take L') ∧
      (G ≠ 4 → N = z.avail_in ∧ SyncSt G (seen ++ bs)) ∧
      (G ≠ 4 → ∃ after,
        C.run (inflateSync (some r)) h = .ok (Z_DATA_ERROR, after) ∧
        after = freeWin
          (setI (r.put (syncAlloc h) (syncAdvance z N)) zs (syncHave s G))
          (syncBuf h) ∧
        Good after r zs (syncAdvance z N) (syncHave s G)) ∧
      (G = 4 → ∃ after,
        C.run (inflateSync (some r)) h = .ok (Z_OK, after) ∧
        after = freeWin
          (syncOkHeap
            (setI (r.put (syncAlloc h) (syncAdvance z N)) zs (syncHave s G))
            r zs (syncAdvance z N) (syncHave s G)) (syncBuf h) ∧
        Good after r zs (syncOkStream (syncAdvance z N) (syncHave s G))
          (syncOkState zs (syncHave s G))) := by
  have fresh := good.fresh
  have allocated := good_syncAlloc good
  have input0 := syncPublicBytes_alloc fresh input
  have distinct := win_ne_buf fresh good.inv.win
  obtain ⟨N, G, used, found, missed, searched, bad, done⟩ :=
    syncTail_run_bytes allocated free input0 atSpec
  have localGet : (syncAlloc h).bytes h.nextBlk = some (syncJunk h) :=
    syncAlloc_bytes_buf h
  refine ⟨N, G, used, found, missed, fun notFound => ?_, fun complete => ?_⟩
  · refine ⟨_, ?_, rfl, ?_⟩
    · show inflateSync (some r) h = _
      rw [inflateSync_split]
      isimp [allocLocalB_run h fresh, stateCheck_live allocated.live,
        allocated.live.zst, good.live.hz, good.live.hs, progress, mode,
        bad notFound,
        freeBuf_run
          (H := setI (r.put (syncAlloc h) (syncAdvance z N)) zs (syncHave s G))
          (nb := h.nextBlk) (a := syncJunk h)
          (by simpa using localGet) (by simp [syncAlloc_ro_buf]), syncBuf]
    · exact good_freeBuf (nb := h.nextBlk) searched
        (fun w window => distinct w (by simpa [syncHave] using window))
  · refine ⟨_, ?_, rfl, ?_⟩
    · have different : ∀ w,
          (syncHave s G).window = .ref (.blk w) 0 → w ≠ h.nextBlk :=
        fun w window => distinct w (by simpa [syncHave] using window)
      have bytes := syncOkHeap_bytes_ne
        (setI (r.put (syncAlloc h) (syncAdvance z N)) zs (syncHave s G))
        r zs (syncAdvance z N) (syncHave s G) different
      show inflateSync (some r) h = _
      rw [inflateSync_split]
      isimp [allocLocalB_run h fresh, stateCheck_live allocated.live,
        allocated.live.zst, good.live.hz, good.live.hs, progress, mode,
        done complete,
        freeBuf_run
          (H := syncOkHeap
            (setI (r.put (syncAlloc h) (syncAdvance z N)) zs (syncHave s G))
            r zs (syncAdvance z N) (syncHave s G))
          (nb := h.nextBlk) (a := syncJunk h)
          (by rw [bytes.1]; simpa using localGet)
          (by rw [bytes.2]; simp [syncAlloc_ro_buf]), syncBuf]
    · have resulting := (syncDone_good searched).1
      exact good_freeBuf (nb := h.nextBlk) resulting
        (fun w window => distinct w
          (by simpa [syncHave] using syncOkState_window_ref window))

/-- The first synchronization call also reads any represented public byte
view after restoring whole bytes from its held-bit buffer. -/
theorem inflateSync_first_bytes {h : Heap} {r : ZStreamRef} {zs : Nat}
    {z : ZStream} {s : InflateState} {bs : List UInt8}
    (good : Good h r zs z s) (free : z.zfree = .fn "zcfree")
    (mode : s.mode ≠ SYNC) (progress : ¬ (z.avail_in = 0 ∧ s.bits < 8))
    (input : Bytes h z.next_in z.avail_in.toNat bs) :
    ∃ N G, N.toNat ≤ z.avail_in.toNat ∧
      (G = 4 → syncPat <:+: syncRestore s ++ bs.take N.toNat ∧
        ∀ L', L' < N.toNat →
          ¬ syncPat <:+: syncRestore s ++ bs.take L') ∧
      (G ≠ 4 → N = z.avail_in ∧ SyncSt G (syncRestore s ++ bs)) ∧
      (G ≠ 4 → ∃ after,
        C.run (inflateSync (some r)) h = .ok (Z_DATA_ERROR, after) ∧
        after = freeWin
          (setI
            (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
              (syncAdvance z N)) zs (syncHave (syncFirstState s) G))
          (syncBuf h) ∧
        Good after r zs (syncAdvance z N) (syncHave (syncFirstState s) G)) ∧
      (G = 4 → ∃ after,
        C.run (inflateSync (some r)) h = .ok (Z_OK, after) ∧
        after = freeWin
          (syncOkHeap
            (setI
              (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
                (syncAdvance z N)) zs (syncHave (syncFirstState s) G))
            r zs (syncAdvance z N) (syncHave (syncFirstState s) G))
          (syncBuf h) ∧
        Good after r zs
          (syncOkStream (syncAdvance z N) (syncHave (syncFirstState s) G))
          (syncOkState zs (syncHave (syncFirstState s) G))) := by
  have fresh := good.fresh
  have allocated := good_syncAlloc good
  have distinct := win_ne_buf fresh good.inv.win
  have localGet : (syncAlloc h).bytes h.nextBlk = some (syncJunk h) :=
    syncAlloc_bytes_buf h
  have localSize : (syncJunk h).size = 4 := by simp [syncJunk]
  have firstGood := good_firstState allocated localGet distinct
  have firstInput := syncPublicBytes_first fresh zs s input
  have firstRun := syncFirst_run
    (H0 := syncAlloc h) (zs := zs) (nb := h.nextBlk)
    (st := s) (A := syncJunk h)
    (by simpa using good.live.hs) localGet localSize (syncAlloc_ro_buf h)
    good.inv.bits
  obtain ⟨N, G, used, found, missed, searched, bad, done⟩ :=
    syncTail_run_bytes firstGood free firstInput (syncAt_restore s)
  have finalLocal :
      (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h)).bytes h.nextBlk =
      some (fillBuf (syncJunk h) (syncHold0 s) (syncNB s)) := by
    simp [syncFirstHeap]
  have finalRo :
      (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h)).ro h.nextBlk = false := by
    simp [syncFirstHeap, syncAlloc_ro_buf]
  refine ⟨N, G, used, found, missed, fun notFound => ?_, fun complete => ?_⟩
  · refine ⟨_, ?_, rfl, ?_⟩
    · show inflateSync (some r) h = _
      rw [inflateSync_split]
      isimp [allocLocalB_run h fresh, stateCheck_live allocated.live,
        allocated.live.zst, good.live.hz, good.live.hs, progress, mode,
        syncBuf, firstRun, bad notFound,
        freeBuf_run
          (H := setI
            (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
              (syncAdvance z N)) zs (syncHave (syncFirstState s) G))
          (nb := h.nextBlk)
          (a := fillBuf (syncJunk h) (syncHold0 s) (syncNB s))
          (by simpa using finalLocal) (by simpa using finalRo)]
    · exact good_freeBuf (nb := h.nextBlk) searched
        (fun w window => distinct w
          (by simpa [syncHave, syncFirstState] using window))
  · refine ⟨_, ?_, rfl, ?_⟩
    · have different : ∀ w,
          (syncHave (syncFirstState s) G).window = .ref (.blk w) 0 →
            w ≠ h.nextBlk :=
        fun w window => distinct w
          (by simpa [syncHave, syncFirstState] using window)
      have bytes := syncOkHeap_bytes_ne
        (setI
          (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
            (syncAdvance z N)) zs (syncHave (syncFirstState s) G))
        r zs (syncAdvance z N) (syncHave (syncFirstState s) G) different
      show inflateSync (some r) h = _
      rw [inflateSync_split]
      isimp [allocLocalB_run h fresh, stateCheck_live allocated.live,
        allocated.live.zst, good.live.hz, good.live.hs, progress, mode,
        syncBuf, firstRun, done complete,
        freeBuf_run
          (H := syncOkHeap
            (setI
              (r.put (syncFirstHeap (syncAlloc h) zs h.nextBlk s (syncJunk h))
                (syncAdvance z N)) zs (syncHave (syncFirstState s) G))
            r zs (syncAdvance z N) (syncHave (syncFirstState s) G))
          (nb := h.nextBlk)
          (a := fillBuf (syncJunk h) (syncHold0 s) (syncNB s))
          (by rw [bytes.1]; simpa using finalLocal)
          (by rw [bytes.2]; simpa using finalRo)]
    · have resulting := (syncDone_good searched).1
      exact good_freeBuf (nb := h.nextBlk) resulting
        (fun w window => distinct w
          (by simpa [syncHave, syncFirstState] using
            syncOkState_window_ref window))

end Zlib.Inflate.Proofs
