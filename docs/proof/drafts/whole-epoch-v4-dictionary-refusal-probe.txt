open Lean Elab Command
run_meta do
  let env ← getEnv
  for moduleName in ["zlibf-whole-epoch-dictionary-transition-v4", "zlibf-whole-epoch-dictionary-request-v4", "zlibf-whole-epoch-entry-refusal-v4"] do
    let some index := env.getModuleIdx? (Name.mkSimple moduleName) | throwError m!"module missing {moduleName}"
    let mut count : Nat := 0
    for (name, info) in env.constants.map₁.toList do
      if env.getModuleIdxFor? name == some index then
        count := count + 1
        if info.isPartial then throwError m!"partial declaration {name}"
        if info.isUnsafe then throwError m!"unsafe declaration {name}"
    logInfo m!"module={moduleName} owned={count} partial=0 unsafe=0"
